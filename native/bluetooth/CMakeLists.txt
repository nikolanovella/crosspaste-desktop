cmake_minimum_required(VERSION 3.20)
project(CrossPasteBluetoothTransport LANGUAGES CXX)
enable_testing()
if(WIN32)
    add_executable(WindowsBluetoothTransport WIN32 windows/WindowsBluetoothTransport.cpp)
    target_compile_features(WindowsBluetoothTransport PRIVATE cxx_std_20)
    target_compile_definitions(WindowsBluetoothTransport PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0601)
    target_link_libraries(WindowsBluetoothTransport PRIVATE ws2_32 shell32)
    if(MSVC)
        set_property(TARGET WindowsBluetoothTransport PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    elseif(MINGW)
        target_link_options(WindowsBluetoothTransport PRIVATE -static -municode)
    endif()
    add_test(NAME WindowsBluetoothTransportSelfTest COMMAND WindowsBluetoothTransport --self-test)
elseif(APPLE)
    enable_language(OBJCXX)
    add_executable(MacosBluetoothTransport macos/MacosBluetoothTransport.mm)
    target_compile_features(MacosBluetoothTransport PRIVATE cxx_std_20)
    set_property(TARGET MacosBluetoothTransport PROPERTY OBJCXX_STANDARD 20)
    target_compile_options(MacosBluetoothTransport PRIVATE -fobjc-arc)
    target_link_libraries(MacosBluetoothTransport PRIVATE "-framework Foundation" "-framework IOBluetooth")
    target_link_options(MacosBluetoothTransport PRIVATE "-Wl,-sectcreate,__TEXT,__info_plist,${CMAKE_CURRENT_SOURCE_DIR}/macos/MacosBluetoothInfo.plist")
    add_test(NAME MacosBluetoothTransportSelfTest COMMAND MacosBluetoothTransport --self-test)
else()
    message(FATAL_ERROR "This Bluetooth adapter supports Windows and macOS only.")
endif()
